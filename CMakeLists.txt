cmake_minimum_required(VERSION 3.20)
project(certskip LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
    OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -c
    "import pyarrow; print(pyarrow.get_include()); print(pyarrow.get_library_dirs()[0])"
    OUTPUT_VARIABLE ARROW_PATHS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(REPLACE "\n" ";" ARROW_PATHS "${ARROW_PATHS}")
list(GET ARROW_PATHS 0 ARROW_INCLUDE)
list(GET ARROW_PATHS 1 ARROW_LIB)
pybind11_add_module(_native cpp/bindings.cpp cpp/certificate.cpp cpp/parquet_reader.cpp cpp/pruning.cpp cpp/selector.cpp)
target_include_directories(_native PRIVATE "${ARROW_INCLUDE}")
foreach(NAME arrow arrow_python parquet)
    file(GLOB LIBS "${ARROW_LIB}/lib${NAME}.so*" "${ARROW_LIB}/lib${NAME}.*dylib")
    list(SORT LIBS)
    list(GET LIBS 0 LIB)
    target_link_libraries(_native PRIVATE "${LIB}")
endforeach()
set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/certskip" BUILD_RPATH "${ARROW_LIB}")
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests cpp/tests/core_tests.cpp cpp/pruning.cpp cpp/selector.cpp)
    target_include_directories(core_tests PRIVATE cpp)
    add_test(NAME core_tests COMMAND core_tests)
endif()
